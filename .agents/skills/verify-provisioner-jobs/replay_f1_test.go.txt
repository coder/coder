package provisionerdserver_test

	"database/sql"
	"encoding/json"
	"testing"
	"time"

	"github.com/google/uuid"
	"github.com/stretchr/testify/require"

	"github.com/coder/coder/v2/coderd/database"
	"github.com/coder/coder/v2/coderd/database/dbgen"
	"github.com/coder/coder/v2/coderd/jobreaper"
	"github.com/coder/coder/v2/coderd/provisionerdserver"
	"github.com/coder/coder/v2/provisionerd/proto"
	"github.com/coder/coder/v2/testutil"
)

// TestFormalReplayF1 replays the F1 path of
// .agents/skills/verify-provisioner-jobs/ProvisionerJobs.tla against Postgres.
// It passes while the bug exists. Copy it into this package to run it, and
// delete the copy afterwards.
func TestFormalReplayF1(t *testing.T) {
	t.Parallel()
	ctx := testutil.Context(t, testutil.WaitLong)
	srv, db, ps, pd := setup(t, false, nil)

	user := dbgen.User(t, db, database.User{})
	tpl := dbgen.Template(t, db, database.Template{CreatedBy: user.ID, OrganizationID: pd.OrganizationID})
	tv := dbgen.TemplateVersion(t, db, database.TemplateVersion{
		CreatedBy: user.ID, OrganizationID: pd.OrganizationID,
		TemplateID: uuid.NullUUID{UUID: tpl.ID, Valid: true}, JobID: uuid.New(),
	})
	ws := dbgen.Workspace(t, db, database.WorkspaceTable{TemplateID: tpl.ID, OwnerID: user.ID, OrganizationID: pd.OrganizationID})

	// Build N-1: finished, with state.
	prevJob := dbgen.ProvisionerJob(t, db, nil, database.ProvisionerJob{
		Type: database.ProvisionerJobTypeWorkspaceBuild, OrganizationID: pd.OrganizationID,
		StartedAt:   sql.NullTime{Time: time.Now().Add(-time.Hour), Valid: true},
		CompletedAt: sql.NullTime{Time: time.Now().Add(-time.Hour), Valid: true},
	})
	prevBuild := dbgen.WorkspaceBuild(t, db, database.WorkspaceBuild{
		JobID: prevJob.ID, WorkspaceID: ws.ID, TemplateVersionID: tv.ID, BuildNumber: 1,
		Transition: database.WorkspaceTransitionStart, Reason: database.BuildReasonInitiator,
	})
	require.NoError(t, db.UpdateWorkspaceBuildProvisionerStateByID(ctx, database.UpdateWorkspaceBuildProvisionerStateByIDParams{
		ID: prevBuild.ID, ProvisionerState: []byte(`"state of build N-1"`), UpdatedAt: time.Now(),
	}))

	// Build N: the daemon acquires its job (DaemonAcquires).
	buildID := uuid.New()
	job := dbgen.ProvisionerJob(t, db, nil, database.ProvisionerJob{
		InitiatorID: user.ID, Type: database.ProvisionerJobTypeWorkspaceBuild,
		Input:          must(json.Marshal(provisionerdserver.WorkspaceProvisionJob{WorkspaceBuildID: buildID})),
		OrganizationID: pd.OrganizationID,
	})
	dbgen.WorkspaceBuild(t, db, database.WorkspaceBuild{
		ID: buildID, JobID: job.ID, WorkspaceID: ws.ID, TemplateVersionID: tv.ID, BuildNumber: 2,
		InitiatorID: user.ID, Transition: database.WorkspaceTransitionStart, Reason: database.BuildReasonInitiator,
	})
	_, err := db.AcquireProvisionerJob(ctx, database.AcquireProvisionerJobParams{
		OrganizationID: pd.OrganizationID, WorkerID: uuid.NullUUID{UUID: pd.ID, Valid: true},
		Types:     []database.ProvisionerType{database.ProvisionerTypeEcho},
		StartedAt: sql.NullTime{Time: time.Now(), Valid: true}, ProvisionerTags: must(json.Marshal(job.Tags)),
	})
	require.NoError(t, err)

	// DaemonFreezes + TimePasses: no heartbeat for 10 minutes.
	require.NoError(t, db.UpdateProvisionerJobByID(ctx, database.UpdateProvisionerJobByIDParams{
		ID: job.ID, UpdatedAt: time.Now().Add(-10 * time.Minute),
	}))

	// ReaperScans + ReaperReapsJob.
	tickCh := make(chan time.Time)
	statsCh := make(chan jobreaper.Stats)
	det := jobreaper.New(ctx, db, ps, testutil.Logger(t), tickCh).WithStatsChannel(statsCh)
	det.Start()
	testutil.RequireSend(ctx, t, tickCh, time.Now())
	stats := testutil.RequireReceive(ctx, t, statsCh)
	det.Close()
	det.Wait()
	require.Equal(t, []uuid.UUID{job.ID}, stats.TerminatedJobIDs)
	reaped, err := db.GetProvisionerJobByID(ctx, job.ID)
	require.NoError(t, err)
	require.Equal(t, database.ProvisionerJobStatusFailed, reaped.JobStatus)
	st, err := db.GetWorkspaceBuildProvisionerStateByID(ctx, buildID)
	require.NoError(t, err)
	t.Logf("after reap: status=%s error=%q state=%s", reaped.JobStatus, reaped.Error.String, st.ProvisionerState)
	// UserStartsNextBuild would copy this state (wsbuilder getState).
	nextBase := string(st.ProvisionerState)

	// DaemonThaws + DaemonHeartbeats: UpdateJob still accepts the daemon.
	upd, err := srv.UpdateJob(ctx, &proto.UpdateJobRequest{JobId: job.ID.String()})
	require.NoError(t, err)
	t.Logf("heartbeat after reap: err=nil canceled=%v", upd.Canceled)

	// DaemonCompletesJob.
	_, err = srv.CompleteJob(ctx, &proto.CompletedJob{
		JobId: job.ID.String(),
		Type: &proto.CompletedJob_WorkspaceBuild_{WorkspaceBuild: &proto.CompletedJob_WorkspaceBuild{
			State: []byte(`"real state of build N"`),
		}},
	})
	require.NoError(t, err)
	after, err := db.GetProvisionerJobByID(ctx, job.ID)
	require.NoError(t, err)
	st2, err := db.GetWorkspaceBuildProvisionerStateByID(ctx, buildID)
	require.NoError(t, err)
	t.Logf("after late CompleteJob: status=%s error=%q state=%s", after.JobStatus, after.Error.String, st2.ProvisionerState)
	t.Logf("next build would have started from %s", nextBase)
	require.Equal(t, database.ProvisionerJobStatusSucceeded, after.JobStatus, "F1: reaped job flipped to succeeded")
	require.NotEqual(t, nextBase, string(st2.ProvisionerState), "F1: build N state changed after reap")
}
